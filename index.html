<script>
const homeScreen = document.getElementById('home-screen');
const galleryScreen = document.getElementById('gallery-screen');
const arScreen = document.getElementById('ar-screen');
let arSystem = null;
function showGallery() {
homeScreen.classList.add('hidden');
arScreen.classList.add('hidden');
galleryScreen.classList.remove('hidden');
}
function goHome() {
galleryScreen.classList.add('hidden');
arScreen.classList.add('hidden');
homeScreen.classList.remove('hidden');
}
function startAR() {
homeScreen.classList.add('hidden');
galleryScreen.classList.add('hidden');
arScreen.classList.remove('hidden');
const sceneEl = document.querySelector('a-scene');
arSystem = sceneEl.systems["mindar-image-system"];
if(arSystem) {
arSystem.start();
}
}
function stopAR() {
if (arSystem) {
arSystem.stop();
}
const video = document.querySelector('#ar-video-1');
if(video) video.pause();
goHome();
}
document.addEventListener("DOMContentLoaded", function() {
const target = document.querySelector('#target-1');
const video = document.querySelector('#ar-video-1');
if(target && video) {
target.addEventListener("targetFound", event => {
console.log("Gambar Dikesan! Memainkan video...");
document.getElementById('ar-instruction').innerHTML = "<p class='text-[#0F766E] font-bold text-sm text-center'>✨ Gambar dijumpai! Menonton cerita... ✨</p>";
document.getElementById('ar-instruction').classList.replace('border-pink-200', 'border-teal-300');
document.getElementById('ar-instruction').classList.add('bg-teal-50');
video.play().catch(e => {
console.log("Auto-play disekat oleh pelayar. Pengguna perlu berinteraksi.");
});
});
target.addEventListener("targetLost", event => {
console.log("Gambar Hilang. Berhenti video.");
document.getElementById('ar-instruction').innerHTML = "<p class='text-[#5C4B51] font-bold text-sm text-center'>✨ Halakan kamera pada gambar (Artwork) untuk melihat keajaiban! ✨</p>";
document.getElementById('ar-instruction').classList.replace('border-teal-300', 'border-pink-200');
document.getElementById('ar-instruction').classList.remove('bg-teal-50');
video.pause();
});
}
});
</script>
